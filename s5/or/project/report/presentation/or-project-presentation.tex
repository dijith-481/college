% 2 — scope and assumptions.
\begin{frame}{A precise, limited question}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Decision}
        Choose green durations for North, East, South, and West:
        \[
          \mathbf g=(g_N,g_E,g_S,g_W),\qquad \sum_i g_i=80\text{ s}.
        \]
        Each 96-second cycle includes 16 seconds of clearance.
      \end{block}
      \begin{block}{Fixed operation}
        \small Four protected phases; two-second service; turn shares of 20/60/20.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Study case}
        Initially empty queues, assumed Poisson arrivals, 12 movement queues, and a 600-second horizon.
        \vspace{0.08cm}

        \textbf{Objective:} minimize expected cumulative waiting. No field measurements are used in this primary experiment.
      \end{block}
  \end{columns}
\end{frame}

% 3 — mathematical model.
\begin{frame}{From arrivals to waiting cost}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \begin{block}{Poisson arrivals}
        Uniform pseudorandom draws are transformed into exponential interarrival gaps:
        \[
          \Delta t=-\frac{\ln(1-U)}{\lambda}.
        \]
        A turning draw assigns each vehicle to a dedicated left, through, or right queue.
      \end{block}
    \column{0.47\textwidth}
      \begin{block}{Queue-area objective}
        \[
          D_T(\mathbf g)=\sum_{m=1}^{12}\int_0^T Q_m(t;\mathbf g)\,dt.
        \]
        \small This is cumulative waiting in vehicle-seconds, including partial waiting still in progress at the horizon.
      \end{block}
  \end{columns}
\end{frame}

% 4 — operations research procedure.
\begin{frame}{Search, then evaluate on fresh traffic}
  \begin{columns}[T,onlytextwidth]
    \column{0.24\textwidth}\metric{1}{Enumerate\\feasible timings}
    \column{0.24\textwidth}\metric{2}{Score on shared\\training arrivals}
    \column{0.24\textwidth}\metric{3}{Select the lowest\\estimated mean}
    \column{0.24\textwidth}\metric{4}{Evaluate on new,\\matched arrivals}
  \end{columns}
  \vspace{0.35cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.58\textwidth}
      \begin{block}{Why simulation?}
        Service is gated by protected green phases and deterministic headways. A continuously serving M/M/1 formula does not represent this operation exactly.
      \end{block}
    \column{0.38\textwidth}
      \begin{block}{Reported measures}
        \small Mean waiting, paired differences, queue profiles, residual backlog, and uncertainty intervals.
      \end{block}
  \end{columns}
\end{frame}
